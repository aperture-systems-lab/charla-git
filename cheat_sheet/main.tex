\subsection*{Crear un repository}
\begin{cheat}[0.34\linewidth]
  \code{git init}        & Crea un repository nuevo \\
  \code{git clone <url>} & Clona un repository existente \\
\end{cheat}

\subsection*{Configuración}
\begin{cheat}[0.5\linewidth]
  \code{git config --global user.name "..."}          & Nombre del autor de cada commit \\
  \code{git config --global user.email "..."}         & Correo del autor, el mismo de GitHub \\
  \code{git config --global init.defaultBranch main}  & Define \code{main} como branch inicial \\
  \code{git config --list}                            & Muestra la configuración vigente \\
\end{cheat}

\begin{cheat}[0.42\linewidth]
  \code{git add <file>}   & Copia el file al staging \\
  \code{git add .}        & Añade todo el contenido del working directory \\
  \code{git reset <file>} & Retira el file del staging \\
  \code{git reset}        & Vacía por completo el staging \\
  \code{git status}       & Muestra el estado de cada file \\
\end{cheat}

\subsection*{Hacer commits}
\begin{cheat}[0.42\linewidth]
  \code{git commit}        & Abre el editor para redactar el mensaje \\
  \code{git commit -m "..."} & Crea el commit con el mensaje indicado \\
\end{cheat}

\subsection*{Trabajo con branches}
\begin{cheat}[0.55\linewidth]
  \code{git switch <branch>} \ /\ \code{git checkout <branch>}     & Cambia de branch \\
  \code{git switch -c <branch>} \ /\ \code{git checkout -b <branch>} & Crea un branch y cambia a él \\
  \code{git branch}           & Lista los branches existentes \\
  \code{git branch -d <branch>} & Elimina un branch ya mergeado \\
  \code{git branch -D <branch>} & Elimina un branch de forma forzada \\
\end{cheat}

\subsection*{Ver diffs}
\begin{cheat}[0.4\linewidth]
  \code{git diff}                  & Working directory frente al staging \\
  \code{git diff --staged}         & Staging frente al último commit \\
  \code{git diff <c1> <c2>}        & Diff entre dos commits \\
  \code{git diff <commit> <file>}  & Un file desde un commit determinado \\
  \code{git show <commit>}         & Diff de un commit frente a su parent \\
\end{cheat}

\subsection*{Log del proyecto}
\begin{cheat}[0.4\linewidth]
  \code{git log}           & Registro con hash, autor, fecha y mensaje \\
  \code{git log --oneline} & Un commit por línea \\
  \code{git log --graph}   & Dibuja el grafo de branches y merges \\
  \code{git log --all}     & Incluye los commits de todos los branches \\
  \code{git log <file>}    & Commits que modificaron un file \\
\end{cheat}

\subsection*{Formas de referirse a un commit}
\begin{cheat}[0.42\linewidth]
  Un branch                     & \code{main} \\
  Un tag                        & \code{v0.1} \\
  Un identificador de commit    & \code{3e887ab} \\
  Un remote branch              & \code{origin/main} \\
  El commit actual              & \code{HEAD} \\
  Tres commits anteriores       & \code{HEAD~3} o \code{HEAD^^^} \\
\end{cheat}

\subsection*{Tags}
\begin{cheat}[0.46\linewidth]
  \code{git tag}                 & Lista los tags existentes \\
  \code{git tag v1.0}            & Marca el commit actual \\
  \code{git tag -a v1.0 -m "..."} & Tag con autor, fecha y mensaje \\
  \code{git tag v1.0 <hash>}     & Marca un commit anterior \\
  \code{git push origin v1.0}    & Publica un tag concreto \\
  \code{git tag -d v1.0}         & Elimina el tag local \\
\end{cheat}

\subsection*{Descarte de cambios}
\begin{cheat}[0.5\linewidth]
  \code{git restore <file>} \ /\ \code{git checkout -- <file>} & Descarta los cambios no añadidos \\
  \code{git restore --staged --worktree <file>} & Descarta los cambios staged y no staged \\
  \code{git reset --hard} & Descarta \textcolor{errc}{\bfseries todo} el trabajo sin commitear \\
  \code{git stash}        & Guarda el trabajo en curso de forma temporal \\
\end{cheat}

\subsection*{Exclusiones con \texttt{.gitignore}}
\begin{cheat}[0.3\linewidth]
  \code{*.csv}       & Cualquier file con extensión \texttt{.csv} \\
  \code{models/}     & La carpeta completa \\
  \code{/notas.txt}  & Únicamente el file de la raíz \\
  \code{**/tmp/}     & En cualquier nivel del árbol \\
  \codeok{!data/muestra.csv} & Excepción, este file sí se incluye \\
\end{cheat}
{\color{black!55}Si el file ya está commiteado, se retira con
\code{git rm --cached <file>}.}

\subsection*{Reescritura del historial}
\begin{cheat}[0.42\linewidth]
  \code{git commit --amend} & Modifica el último commit \\
  \code{git reflog}         & Recupera commits que quedaron sin branch \\
\end{cheat}

\subsection*{Elección del comando para deshacer}
\begin{cheat}[0.55\linewidth]
  Errata en el último commit aún sin push     & \code{git commit --amend} \\
  Deshacer commits sin push                   & \code{git reset <flag> HEAD~1} \\
  Deshacer un commit ya publicado             & \code{git revert <hash>} \\
  Descartar los cambios de un file            & \code{git restore <file>} \\
  Recuperar un file de una versión anterior   & \code{git checkout <hash> -- <file>} \\
  Recuperar commits perdidos                  & \code{git reflog} \\
\end{cheat}

\noindent\begin{minipage}{\linewidth}
\begin{topics}
  \item \textbf{Fast-forward} desplaza el branch sin crear un commit.
  \item \textbf{Merge commit} enlaza dos historias con un commit de dos parents.
  \item \textbf{Rebase} reaplica los commits y genera hashes nuevos.
  \item \textbf{Cherry-pick} copia un commit concreto en otro branch.
  \item \textbf{Squash merge} concentra todos los commits en uno.
\end{topics}
\begin{cheat}[0.42\linewidth]
  \code{git merge <branch>}         & Mergea el branch indicado \\
  \code{git merge --no-ff <branch>} & Fuerza la creación de un merge commit \\
  \code{git rebase <branch>}        & Reaplica los commits sobre el otro branch \\
  \code{git cherry-pick <hash>}     & Copia un único commit \\
  \code{git merge --squash <branch>} & Prepara el merge en un solo commit \\
\end{cheat}
\end{minipage}

\noindent\begin{minipage}{\linewidth}
\subsection*{Conflictos}
Se producen cuando dos branches modifican las mismas líneas de un file.
\begin{term}
@-<<<<<<< HEAD
@+    return m
@>=======
@*    return round(m, 2)
@*>>>>>>> redondeo
\end{term}
La resolución consiste en editar el file, eliminar los marcadores y
commitear con \code{git add} y \code{git commit}. El merge se cancela con
\code{git merge --abort}.
\end{minipage}

\subsection*{Cerrar un branch}
\begin{cheat}[0.42\linewidth]
  \code{git branch --merged}    & Lista los branches ya mergeados \\
  \code{git branch -d <branch>} & Elimina el branch y conserva los commits \\
  \code{git branch -D <branch>} & Elimina el branch de forma forzada \\
\end{cheat}

\subsection*{Remote repository}
\begin{cheat}[0.4\linewidth]
  \code{git remote add <name> <url>}  & Enlaza el repository local con GitHub \\
  \code{git push origin main}         & Hace push de \code{main} al remote \code{origin} \\
  \code{git push -u origin <branch>}  & Registra el upstream la primera vez \\
  \code{git push}                     & Hace push al upstream ya registrado \\
  \code{git push --force-with-lease}  & Reescritura forzada con verificación previa \\
  \code{git push --tags}              & Hace push de los tags \\
\end{cheat}

\subsection*{Fetch y pull}
\begin{cheat}[0.32\linewidth]
  \code{git fetch origin main} & Descarga sin modificar el branch local \\
  \code{git pull}              & Equivale a \code{fetch} y \code{merge} \\
  \code{git pull --rebase}     & Equivale a \code{fetch} y \code{rebase} \\
\end{cheat}

\noindent\begin{minipage}{\linewidth}
\subsection*{Fork y pull request}
\begin{topics}
  \item \textbf{Fork} crea una copia del repository en la cuenta propia.
  \item \textbf{Pull request} solicita la incorporación de los cambios.
\end{topics}
\end{minipage}

\noindent\begin{minipage}{\linewidth}
\subsection*{Redacción de mensajes de commit}
El mensaje es la única parte del commit redactada por el autor y determina la
legibilidad del historial.

\noindent{\footnotesize
\rowcolors{1}{primary!6}{white}%
\begin{tabularx}{\linewidth}{@{\hspace{3pt}}c@{\hspace{6pt}}L@{\hspace{3pt}}}
  \no & \code{cambios} · \code{update} · \code{asdf} · \code{ahora si}\newline
        {\color{black!55}No indica qué cambió ni por qué} \\
  \si & \code{fix: entrena sin el conjunto de prueba} ·
        \code{feat: anade el cargador de CSV}\newline
        {\color{black!55}Indica qué cambió y por qué} \\
\end{tabularx}}
\end{minipage}

\subsection*{Formato Conventional Commits 1.0.0}
\begin{center}
{\ttfamily\textcolor{okc}{<tipo>}\textcolor{stgc}{[(ámbito)]}: <descripción>}
\end{center}
\noindent{\footnotesize
\begin{tabularx}{\linewidth}{@{}L@{\hspace{5pt}}L@{\hspace{5pt}}L@{}}
  \textcolor{okc}{\bfseries tipo} clase de cambio &
  \textcolor{stgc}{\bfseries ámbito} parte afectada, opcional &
  \textbf{descripción} acción en presente \\
\end{tabularx}}\par\smallskip
{\footnotesize El \textbf{cuerpo} es opcional, se añade con un segundo
\code{-m} y expone el motivo del cambio.}
\begin{term}
git commit -m "fix(modelo): entrena sin el conjunto de prueba" \
  -m "Ajustaba con el conjunto de prueba y eso sesgaba la métrica."
\end{term}

\subsection*{Tipos}
\noindent{\footnotesize
\rowcolors{1}{primary!6}{white}%
\begin{tabularx}{\linewidth}{@{\hspace{3pt}}l@{\hspace{6pt}}L@{\hspace{6pt}}l@{\hspace{6pt}}L@{\hspace{3pt}}}
  \code{feat}     & Funcionalidad nueva                        & \code{perf}  & Mejora de rendimiento \\
  \code{fix}      & Corrección de un error                     & \code{test}  & Pruebas \\
  \code{docs}     & Documentación                              & \code{ci}    & Integración continua \\
  \code{style}    & Formato sin cambio de código               & \code{chore} & Dependencias y limpieza \\
  \code{refactor} & Reestructuración sin cambio funcional      &              & \\
\end{tabularx}}

\subsection*{Buenas prácticas}
\begin{topics}
  \item \textbf{Modo imperativo} en la descripción, «añade» en lugar de «añadido».
  \item \textbf{Título breve}, de unos 50 caracteres.
  \item \textbf{Un commit por cada cambio lógico}.
  \item \textbf{Referencia al issue} correspondiente.
  \item \textbf{Línea en blanco} entre el título y el cuerpo.
\end{topics}
